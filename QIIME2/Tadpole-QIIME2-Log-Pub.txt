QIIME2 terminal commands


cd BaseSpace/Tadpoles/All-seq/
source activate qiime2-2023.5

qiime tools import --type 'SampleData[PairedEndSequencesWithQuality]' --input-path Tadpole_Manifest.csv --output-path paired-end-demux.qza --input-format PairedEndFastqManifestPhred33

qiime demux summarize --i-data paired-end-demux.qza --o-visualization paired-end-demux.qzv

qiime dada2 denoise-paired --i-demultiplexed-seqs paired-end-demux.qza --o-table table --o-representative-sequences rep-seqs --p-n-threads 0 --verbose --p-trunc-len-f 220 --p-trunc-len-r 220 --p-trim-left-f 19 --p-trim-left-r 20 --o-denoising-stats denoising-stats

qiime feature-table summarize --i-table table.qza --o-visualization table.qzv --m-sample-metadata-file Tadpole_Metadata_All.txt

qiime feature-table tabulate-seqs --i-data rep-seqs.qza --o-visualization rep-seqs.qzv

qiime alignment mafft --i-sequences rep-seqs.qza --o-alignment aligned-rep-seqs.qza

qiime alignment mask --i-alignment aligned-rep-seqs.qza --o-masked-alignment masked-alignment-rep-seqs.qza

qiime phylogeny fasttree --i-alignment masked-alignment-rep-seqs.qza --o-tree unrooted-tree.qza

qiime phylogeny midpoint-root --i-tree unrooted-tree.qza --o-rooted-tree rooted-tree.qza

qiime feature-classifier classify-sklearn --i-classifier silva-138-99-515-806-nb-classifier.qza --i-reads rep-seqs.qza --o-classification taxonomy.qza

qiime metadata tabulate --m-input-file taxonomy.qza --o-visualization taxonomy.qzv

qiime taxa filter-table --i-table table.qza --i-taxonomy taxonomy.qza --p-exclude archaea,chloroplast,mitochondria --o-filtered-table table-sans-a-m-c.qza

qiime metadata tabulate --m-input-file table-sans-a-m-c.qza --o-visualization table-sans-a-m-c.qzv

qiime feature-table summarize --i-table table-sans-a-m-c.qza --o-visualization table-sans-a-m-c --m-sample-metadata-file Tadpole_Metadata_All.txt
